- To display only directories in a folder
	ls -d */
	or 
	ls -ld */ 	# to display as a list
	ls -l --time=birth # to display creation date/time
	ls -lt # To sort by modification time
	ls -ltr # To sort by modification time in reverse order

- To delete one file
	rm filename.txt

	rm -i filename.txt (interactive flag)

- To use VIM to edit a text file
	vim practice.txt
	i → write three lines → Esc
	dd to delete a line, u to undo
	/line to search
	:wq

	:w 	Save
	:q	Quit (fails if unsaved changes)
	:wq	Save and quit
	:q!	Quit without saving
	:x	Save and quit (like :wq if changed)

- To copy files and entire directories
	cp source.txt destination.txt
	cp source.txt /path/to/directory/ (to another directory)
	cp -p source.txt destination.txt (to preserve file attributes, like timestamps, permissions, etc.)
	cp -i source.txt destination.txt (Prompt before overwriting an existing file)
	cp -r /path/to/source_folder /path/to/destination_folder (copy an entire directory recursively)

- To rename a file
	mv old_filename.txt new_filename.txt
	mv -i old_filename.txt new_filename.txt (Prompt before overwriting an existing file with the new name)
	mv -n old_filename.txt new_filename.txt (Avoid overwriting entirely: Use -n to ensure an existing file is never overwritten)
	mv "my old file.txt" "my new file.txt" (Files with spaces: Enclose filenames containing spaces in quotes)
